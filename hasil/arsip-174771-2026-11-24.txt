<!-- wp:paragraph -->
<p><strong>Lirik Lagu Batak: Unang Ahu Solsoli</strong></p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p>Unang au sai solsoli hasian<br>Unang au sai tangisi</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p>Da na au naso ra marningot janji<br>Alai hodo hasian na pittor muli</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p>Dung malala rohakki<br>Roma ho sumolsol bagi</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p>Tung saribu hali ro ho manolsoli<br>Au do sitaon na hassit I</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p>Ditaoni ajibata do ito<br>Cincin mi naung hubolokkon</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p>Unang sai huingot ingoti ito<br>Gabe pa sunggul sunggul sidangolonki</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p>Tu tao nabagas I<br>Hupatolhas lungun ki</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p>Laos disido husumpahon hasian<br>Nasojadi martunangan</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p>Ito hasian unang sai sunggul sungguli be naung salpui<br>Ito anggo holong dohononmu holan tu ho do</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p>Ganjang ma umur mi<br>Leleng hita mangolu hasian</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p>Anggo au hasian boi berengon mu<br>Sahat rodi nalao mate hasian<br>Tung naso jadi au mangoli</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p><strong>Lirik Lagu Batak: Unang Ahu Solsoli</strong></p>
<!-- /wp:paragraph -->
